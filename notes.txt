import com.documents4j.api.*;
import com.documents4j.job.LocalConverter;

import java.io.*;
import java.util.concurrent.TimeUnit;

public class PptxToPdf {

    public static void convert(String pptxPath, String pdfPath) throws Exception {
        IConverter converter = LocalConverter.builder()
            .workerPool(1, 4, 5, TimeUnit.SECONDS)
            .build();

        try (InputStream in = new FileInputStream(pptxPath);
             OutputStream out = new FileOutputStream(pdfPath)) {

            boolean success = converter.convert(in)
                .as(DocumentType.MS_POWERPOINT)
                .to(out)
                .as(DocumentType.PDF)
                .execute();

            if (!success) {
                throw new RuntimeException("Conversion failed");
            }
        } finally {
            converter.shutDown();
        }
    }

    public static void main(String[] args) throws Exception {
        convert("presentation.pptx", "output.pdf");
    }
}

import com.documents4j.api.*;
import com.documents4j.job.LocalConverter;
import org.springframework.core.io.ByteArrayResource;
import org.springframework.http.*;
import org.springframework.web.bind.annotation.*;

import java.io.*;
import java.nio.file.*;
import java.util.concurrent.TimeUnit;

@RestController
@RequestMapping("/pptx")
public class PptxViewerController {

    @GetMapping("/view")
    public ResponseEntity<ByteArrayResource> viewPptx(
            @RequestParam String filePath) {

        Path path = Path.of(filePath);

        // Basic security check - prevent path traversal
        if (!path.isAbsolute() || !Files.exists(path) || !filePath.endsWith(".pptx")) {
            return ResponseEntity.badRequest().build();
        }

        try {
            byte[] pdfBytes = convertToPdf(path.toFile());
            ByteArrayResource resource = new ByteArrayResource(pdfBytes);

            return ResponseEntity.ok()
                    .contentType(MediaType.APPLICATION_PDF)
                    .contentLength(pdfBytes.length)
                    .header(HttpHeaders.CONTENT_DISPOSITION, "inline; filename=\""
                            + path.getFileName().toString().replace(".pptx", ".pdf") + "\"")
                    .body(resource);

        } catch (Exception e) {
            return ResponseEntity.internalServerError().build();
        }
    }

    private byte[] convertToPdf(File pptxFile) throws Exception {
        ByteArrayOutputStream out = new ByteArrayOutputStream();

        IConverter converter = LocalConverter.builder()
                .workerPool(1, 4, 5, TimeUnit.SECONDS)
                .build();

        try (InputStream in = new FileInputStream(pptxFile)) {
            boolean success = converter.convert(in)
                    .as(DocumentType.MS_POWERPOINT)
                    .to(out)
                    .as(DocumentType.PDF)
                    .execute();

            if (!success) {
                throw new RuntimeException("Conversion failed for: " + pptxFile.getName());
            }
        } finally {
            converter.shutDown();
        }

        return out.toByteArray();
    }
}

import org.springframework.stereotype.Component;
import java.util.concurrent.*;

@Component
public class PptxViewerController {

    // Cache: filePath -> PDF bytes
    private final ConcurrentHashMap<String, byte[]> cache = new ConcurrentHashMap<>();

    @GetMapping("/view")
    public ResponseEntity<ByteArrayResource> viewPptx(@RequestParam String filePath) {

        Path path = Path.of(filePath);
        if (!path.isAbsolute() || !Files.exists(path) || !filePath.endsWith(".pptx")) {
            return ResponseEntity.badRequest().build();
        }

        try {
            byte[] pdfBytes = cache.computeIfAbsent(filePath, key -> {
                try {
                    return convertToPdf(path.toFile());
                } catch (Exception e) {
                    throw new RuntimeException(e);
                }
            });

            ByteArrayResource resource = new ByteArrayResource(pdfBytes);
            return ResponseEntity.ok()
                    .contentType(MediaType.APPLICATION_PDF)
                    .contentLength(pdfBytes.length)
                    .header(HttpHeaders.CONTENT_DISPOSITION, "inline; filename=\""
                            + path.getFileName().toString().replace(".pptx", ".pdf") + "\"")
                    .body(resource);

        } catch (Exception e) {
            return ResponseEntity.internalServerError().build();
        }
    }

    // ... same convertToPdf method as above
}

import org.springframework.core.io.ByteArrayResource;
import org.springframework.http.*;
import org.springframework.web.bind.annotation.*;

import java.nio.file.*;

@RestController
@RequestMapping("/files")
public class FileDownloadController {

    private static final Map<String, MediaType> ALLOWED_TYPES = Map.of(
        ".pptx", MediaType.parseMediaType("application/vnd.openxmlformats-officedocument.presentationml.presentation"),
        ".xlsx", MediaType.parseMediaType("application/vnd.openxmlformats-officedocument.spreadsheetml.sheet")
    );

    @GetMapping("/download")
    public ResponseEntity<ByteArrayResource> download(@RequestParam String filePath) throws IOException {

        Path path = Path.of(filePath);
        String fileName = path.getFileName().toString();

        // Security + type check
        String ext = fileName.substring(fileName.lastIndexOf('.')).toLowerCase();
        if (!path.isAbsolute() || !Files.exists(path) || !ALLOWED_TYPES.containsKey(ext)) {
            return ResponseEntity.badRequest().build();
        }

        byte[] data = Files.readAllBytes(path);

        return ResponseEntity.ok()
                .contentType(ALLOWED_TYPES.get(ext))
                .contentLength(data.length)
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"" + fileName + "\"")
                .body(new ByteArrayResource(data));
    }
}

/// Stream data instead of loading in memory

@GetMapping("/download")
public ResponseEntity<InputStreamResource> download(@RequestParam String filePath) throws IOException {

    Path path = Path.of(filePath);
    String fileName = path.getFileName().toString();
    String ext = fileName.substring(fileName.lastIndexOf('.')).toLowerCase();

    if (!path.isAbsolute() || !Files.exists(path) || !ALLOWED_TYPES.containsKey(ext)) {
        return ResponseEntity.badRequest().build();
    }

    return ResponseEntity.ok()
            .contentType(ALLOWED_TYPES.get(ext))
            .contentLength(Files.size(path))
            .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"" + fileName + "\"")
            .body(new InputStreamResource(Files.newInputStream(path)));
}
